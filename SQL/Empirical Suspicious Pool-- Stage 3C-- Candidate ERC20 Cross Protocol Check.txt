--
-- PURPOSE:
--
-- Retrieve every ERC-20 Transfer event occurring inside the
-- four candidate transactions.
--
-- We will inspect the receiving addresses to determine

SELECT

    evt_block_time,
    evt_block_number,

    evt_tx_hash,
    evt_index,

    contract_address AS token_address,

    "from",
    "to",

    value

FROM erc20_ethereum.evt_transfer

WHERE evt_tx_hash IN (

    -- Candidate 1 — Swap 17
    0xddf048be43bb9ba15174234493b0b91dfda8ba268246191921556bd82adef228,

    -- Candidate 1 — Swap 18
    0xaab69a34d189fa3f7062b67b11ae722dffde736ea97bc0790cd0d475ace6e7bb,

    -- Candidate 2 — Swap 32
    0xad2a6f82167b2a62e19ead1af47f57635a9c6cb37128a10d2aa61fb8318f03c4,

    -- Candidate 2 — Swap 33
    0xdd990ce43fec4d3ad42f5edc91839162e76c376840c0050d0c243e3c47ff8c94
)

ORDER BY
    evt_block_number,
    evt_tx_hash,
    evt_index;
