contract_creation AS (

    SELECT
        address,
        MIN(block_time) AS contract_deployed_at,
        MIN(block_number) AS contract_deployed_block

    FROM ethereum.creation_traces

    WHERE address IN (
        SELECT address
        FROM intermediaries
    )

    GROUP BY address
)

SELECT

    i.address,

    -- Dune ownership identification
    oa.owner_key,
    oa.custody_owner,
    oa.account_owner,
    oa.contract_name,
    oa.eoa,
    oa.source,
    oa.source_website,
    oa.source_evidence,

    -- Entity information
    od.name AS entity_name,
    od.primary_category,
    od.category_tags,

    -- Known CEX address?
    ca.cex_name AS cex_name,

    -- Known CEX deposit address?
    cda.cex_name AS deposit_cex_name,
    cda.deposit_first_block_time,

    -- Smart-contract historical information
    cc.contract_deployed_at,
    cc.contract_deployed_block,

    CASE
        WHEN cc.contract_deployed_at IS NOT NULL
         AND cc.contract_deployed_at <=
             TIMESTAMP '2020-12-02 19:27:15'
        THEN TRUE
        ELSE FALSE
    END AS contract_existed_by_candidate_time

LEFT JOIN labels.owner_addresses oa
    ON i.address = oa.address
   AND oa.blockchain = 'ethereum'

LEFT JOIN labels.owner_details od
    ON oa.owner_key = od.owner_key

LEFT JOIN cex.addresses ca
    ON i.address = ca.address
   AND ca.blockchain = 'ethereum'

LEFT JOIN cex.deposit_addresses cda
    ON i.address = cda.address
   AND cda.blockchain = 'ethereum'

LEFT JOIN contract_creation cc
    ON i.address = cc.address

ORDER BY i.address;
