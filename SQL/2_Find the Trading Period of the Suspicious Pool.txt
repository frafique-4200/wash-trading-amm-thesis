SELECT
    COUNT(*) AS number_of_trades,
    MIN(block_time) AS first_trade,
    MAX(block_time) AS last_trade
FROM dex.trades
WHERE blockchain = 'ethereum'
  AND project = 'uniswap'
  AND version = '2'
  AND project_contract_address = 0xe75ffcb48f08da718f8955fac8289eefe96b7013;
