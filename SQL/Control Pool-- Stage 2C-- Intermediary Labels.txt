-- STAGE 2C — INTERMEDIARY LABEL CHECK
--
-- PURPOSE
-- ============================================================
--
-- Stage 2B identified addresses that create possible
-- 2-hop ETH-transfer relationships between candidate
-- transaction initiators:
--
--     Sender A
--        |
--     Intermediary
--        |
--     Sender B
--
-- However, a public service address such as a DEX, CEX,
-- aggregator, bridge, lending protocol, etc. can connect
-- otherwise unrelated users.
--
-- Therefore, before accepting a 2-hop relationship as
-- evidence of common entity control, we check whether each
-- intermediary is already identified as a public/service
-- address in Dune's owner-address labels.
--
-- This is the SAME label-check logic used for the suspicious
-- pool. Only the intermediary-address list is different.
--
-- ============================================================

-- ============================================================
-- STEP 2 — CHECK DUNE OWNER-ADDRESS LABELS
-- ============================================================
--
-- labels.owner_addresses provides higher-level ownership /
-- custody information for known Ethereum addresses.
--
-- A populated owner_key or custody_owner can indicate that
-- an intermediary belongs to an identifiable service.
--
-- We use a LEFT JOIN because we want ALL 40 intermediary
-- addresses returned, including addresses for which Dune
-- currently has no owner label.
--
-- An unlabeled address is NOT automatically treated as a
-- private address. It simply means that this specific Dune
-- label table did not identify it.


SELECT

    -- Intermediary discovered in Stage 2B.
    i.address,


    -- Dune owner identifier, when available.
    oa.owner_key,


    -- Custodial/service owner, when available.
    oa.custody_owner


FROM intermediary_addresses i


LEFT JOIN labels.owner_addresses oa

    ON i.address = oa.address


-- ============================================================
-- STEP 3 — SORT FOR EASY REVIEW
-- ============================================================

ORDER BY

    i.address;
