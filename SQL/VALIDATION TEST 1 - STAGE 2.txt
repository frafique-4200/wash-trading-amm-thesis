-- VALIDATION TEST 1 - STAGE 2
-- Verify the direct ETH-transfer relationship explicitly
-- reported by the paper.
-- ============================================================

SELECT
    hash,
    block_number,
    block_time,
    "from",
    "to",
    value,
    value / 1e18 AS eth_amount
FROM ethereum.transactions
WHERE hash IN (
    0xd09120868a1b8e318c3f051d719ea6ae406936e9d0ecbda278e2d151e66fe141,
    0xf8e463d0cf7c391d319eafee08cac2d6e2c84debb11bbc6ab76fde444a76ed34,
    0xf8c933ee137cb9cf289c5bbc5f6a98fea75fc6efb69ef91a904258a2acce4536
)
ORDER BY block_number;
