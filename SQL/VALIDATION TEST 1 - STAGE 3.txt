-- ============================================================
-- VALIDATION TEST 1 - STAGE 3
-- Inspect all decoded DEX trades made inside the two
-- transactions from the paper's known wash-trading example.
--
-- PURPOSE:
-- Check whether the transactions also trade through other
-- liquidity pools, which could indicate arbitrage/asset
-- conversion rather than wash trading.
-- ============================================================

SELECT
    block_time,
    block_number,
    tx_hash,
    project,
    version,
    project_contract_address AS pool,
    token_sold_symbol,
    token_bought_symbol,
    token_sold_amount,
    token_bought_amount
FROM dex.trades
WHERE blockchain = 'ethereum'
  AND tx_hash IN (
      0x78d1713837aae1d0995383550900ef33efc321e8870bdba7fbf5e398f7484eec,
      0xa28546bd2203675f0dc27cad7a45549f4c71dc1c4cfaa82a485d3eafcc7125ac
  )
ORDER BY block_number, evt_index;
