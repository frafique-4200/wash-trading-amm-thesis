-- =====================================================================
-- QUERY 13
-- Identify Tokens and Token Decimals for the Four Additional Pools
-- =====================================================================
--
-- PURPOSE
-- -------
-- This query identifies token0 and token1 for each of the two suspicious
-- pools and their matched control pools. It also retrieves each token's
-- symbol and decimal count.
--
-- WHY TOKEN DECIMALS ARE REQUIRED
-- -------------------------------
-- Uniswap stores token amounts as whole-number blockchain units.
-- To obtain the actual token amount, we later calculate:
--
--     adjusted amount = raw amount / (10 ^ token decimals)
--
-- This information will therefore be used when preprocessing the raw
-- Swap-event files obtained from Queries 12A–12D.
--
-- EXPECTED OUTPUT
-- ---------------
-- One row per pool, giving four rows in total.
-- =====================================================================


-- ---------------------------------------------------------------------
-- STEP 1: Define the four pools included in the experiment.
--
-- display_order controls the order in which the results are shown.
-- comparison_name identifies the suspicious/control comparison.
-- pool_role distinguishes suspicious pools from matched controls.
-- pool_address is the Uniswap V2 pair-contract address.
-- ---------------------------------------------------------------------
WITH selected_pools (
    display_order,
    comparison_name,
    pool_role,
    pool_address
) AS (
    VALUES

        -- ECO-USDT suspicious pool
        (
            1,
            'ECO-USDT comparison',
            'suspicious',
            0x4dfcb1482878ededccd6cc19f9d043cdd56ffd06
        ),

        -- Matched reference-negative control for ECO-USDT
        (
            2,
            'ECO-USDT comparison',
            'control',
            0xa14dfbaa23ee8e8b69878381f8fd1d0bc502c043
        ),

        -- wCELO suspicious pool
        (
            3,
            'wCELO comparison',
            'suspicious',
            0x8d1ca95559abed542ef4402a252974221b5e1036
        ),

        -- Matched reference-negative control for wCELO
        (
            4,
            'wCELO comparison',
            'control',
            0x669fbbd3612910ff1394f657900fbb8f335d044d
        )
),


-- ---------------------------------------------------------------------
-- STEP 2: Identify token0 and token1 for each pool.
--
-- When a Uniswap V2 pool is created, the Factory emits a PairCreated
-- event. That event records:
--
--     pair   = address of the liquidity pool
--     token0 = address of the pool's first token
--     token1 = address of the pool's second token
--
-- Joining through the pool address gives the correct token0/token1
-- ordering used by amount0In, amount0Out, amount1In and amount1Out in
-- the raw Swap-event files.
-- ---------------------------------------------------------------------
pool_tokens AS (
    SELECT
        p.display_order,
        p.comparison_name,
        p.pool_role,
        p.pool_address,
        f.token0 AS token0_address,
        f.token1 AS token1_address
    FROM selected_pools AS p
    LEFT JOIN uniswap_v2_ethereum.Factory_evt_PairCreated AS f
        ON f.pair = p.pool_address
),


-- ---------------------------------------------------------------------
-- STEP 3: Add the symbol and decimal count of each token.
--
-- tokens.erc20 contains Ethereum ERC-20 token metadata.
--
-- LEFT JOIN is used so that the pool still appears in the output even
-- if Dune does not have a symbol or decimal record for one token.
-- A missing value would appear as NULL and would require investigation.
-- ---------------------------------------------------------------------
pool_metadata AS (
    SELECT
        p.display_order,
        p.comparison_name,
        p.pool_role,
        p.pool_address,

        p.token0_address,
        t0.symbol AS token0_symbol,
        t0.decimals AS token0_decimals,

        p.token1_address,
        t1.symbol AS token1_symbol,
        t1.decimals AS token1_decimals

    FROM pool_tokens AS p

    LEFT JOIN tokens.erc20 AS t0
        ON t0.blockchain = 'ethereum'
       AND t0.contract_address = p.token0_address

    LEFT JOIN tokens.erc20 AS t1
        ON t1.blockchain = 'ethereum'
       AND t1.contract_address = p.token1_address
)


-- ---------------------------------------------------------------------
-- STEP 4: Return one clearly labelled row for each pool.
-- ---------------------------------------------------------------------
SELECT
    comparison_name,
    pool_role,
    pool_address,

    token0_address,
    token0_symbol,
    token0_decimals,

    token1_address,
    token1_symbol,
    token1_decimals

FROM pool_metadata
ORDER BY display_order;
