-- =====================================================================
-- QUERY 12B
-- GET RAW SWAP EVENTS FOR THE ECO-USDT MATCHED CONTROL POOL
-- =====================================================================
--
-- PURPOSE
--
-- Download every raw Swap event belonging to the selected reference-
-- negative control pool during the ECO-USDT suspicious pool's exact
-- observation window.
--
-- Control-pool address:
--   0xa14dfbaa23ee8e8b69878381f8fd1d0bc502c043
--
-- Expected number of rows:
--   7,815
--
-- This pool was reconstructed from the paper-period non-abnormal
-- Uniswap V2 set and selected by raw-Swap-count similarity.
-- =====================================================================

SELECT
    contract_address,

    evt_tx_hash,
    evt_tx_from,
    evt_tx_to,
    evt_tx_index,

    evt_index,
    evt_block_time,
    evt_block_number,
    evt_block_date,

    amount0In,
    amount0Out,
    amount1In,
    amount1Out,

    sender,
    "to"

FROM uniswap_v2_ethereum.pair_evt_swap

WHERE contract_address =
    0xa14dfbaa23ee8e8b69878381f8fd1d0bc502c043

  -- Use the ECO-USDT suspicious pool's observation window.
  AND evt_block_time >= TIMESTAMP '2020-11-09 12:13:04'

  AND evt_block_time <= TIMESTAMP '2022-08-23 10:54:06'

  -- Enforce the AMM paper's block range.
  AND evt_block_number BETWEEN 10000835 AND 15400000

ORDER BY
    evt_block_number,
    evt_index;
