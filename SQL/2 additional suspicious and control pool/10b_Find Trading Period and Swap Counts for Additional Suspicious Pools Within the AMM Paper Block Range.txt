-- =====================================================================
-- QUERY 10b — REVISED
-- FIND THE TRADING PERIOD AND SWAP COUNTS FOR THE TWO ADDITIONAL
-- SUSPICIOUS POOLS WITHIN THE AMM PAPER'S DATA-COLLECTION RANGE
-- =====================================================================
--
-- PURPOSE
--
-- For each additional suspicious pool, this query calculates:
--
--   1. Number of decoded trades in dex.trades
--   2. First and last decoded-trade timestamps
--   3. Number of raw Uniswap V2 Swap events
--   4. First and last raw-Swap timestamps
--
-- Only activity within the AMM paper's data-collection period is used:
--
--   Starting block: 10,000,835
--   Ending block:   15,400,000
--
-- According to the paper, this corresponds approximately to:
--
--   4 May 2020 through 24 August 2022
--
-- The SQL keyword BETWEEN includes both boundary blocks.
-- =====================================================================


-- =====================================================================
-- STEP 1 — DEFINE THE TWO SUSPICIOUS POOLS
-- =====================================================================
--
-- This temporary table contains a readable name and the Ethereum
-- contract address of each selected Uniswap V2 pool.
-- =====================================================================

WITH target_pools AS (

    SELECT
        'BOND-USDC' AS pool_name,
        0x6591c4bcd6d7a1eb4e537da8b78676c1576ba244
            AS pool_address

    UNION ALL

    SELECT
        'WHOLE' AS pool_name,
        0xd4d090ac5b3729159f2f69fa1e8800c243ae5b58
            AS pool_address
),


-- =====================================================================
-- STEP 2 — COUNT THE DECODED TRADES
-- =====================================================================
--
-- dex.trades is Dune's standardized DEX-trade table.
--
-- The conditions require each record to:
--
--   - belong to Ethereum;
--   - come from Uniswap V2;
--   - belong to one of the two target pools;
--   - fall within blocks 10,000,835–15,400,000.
--
-- COUNT(d.tx_hash)
--     Counts the decoded trade records.
--
-- MIN(d.block_time)
--     Finds the first decoded trade within the paper's block range.
--
-- MAX(d.block_time)
--     Finds the last decoded trade within the paper's block range.
-- =====================================================================

decoded_trades AS (

    SELECT
        p.pool_name,
        p.pool_address,

        COUNT(d.tx_hash)
            AS number_of_decoded_trades,

        MIN(d.block_time)
            AS first_decoded_trade,

        MAX(d.block_time)
            AS last_decoded_trade

    FROM target_pools AS p

    LEFT JOIN dex.trades AS d
        ON d.project_contract_address = p.pool_address
       AND d.blockchain = 'ethereum'
       AND d.project = 'uniswap'
       AND d.version = '2'

       -- Apply the AMM paper's data-collection range.
       AND d.block_number BETWEEN 10000835 AND 15400000

    GROUP BY
        p.pool_name,
        p.pool_address
),


-- =====================================================================
-- STEP 3 — COUNT THE RAW UNISWAP V2 SWAP EVENTS
-- =====================================================================
--
-- pair_evt_swap contains the raw Swap events emitted by Uniswap V2
-- liquidity-pool contracts.
--
-- Each matching row represents one Swap event.
--
-- COUNT(s.evt_tx_hash)
--     Counts the raw Swap events.
--
-- MIN(s.evt_block_time)
--     Finds the first raw Swap within the paper's block range.
--
-- MAX(s.evt_block_time)
--     Finds the last raw Swap within the paper's block range.
--
-- These raw Swap events are the records required by our replication
-- and WTEYE-adaptation pipelines.
-- =====================================================================

raw_swaps AS (

    SELECT
        p.pool_name,
        p.pool_address,

        COUNT(s.evt_tx_hash)
            AS number_of_raw_swaps,

        MIN(s.evt_block_time)
            AS first_raw_swap,

        MAX(s.evt_block_time)
            AS last_raw_swap

    FROM target_pools AS p

    LEFT JOIN uniswap_v2_ethereum.pair_evt_swap AS s
        ON s.contract_address = p.pool_address

       -- Apply the same AMM paper data-collection range.
       AND s.evt_block_number BETWEEN 10000835 AND 15400000

    GROUP BY
        p.pool_name,
        p.pool_address
)


-- =====================================================================
-- STEP 4 — COMBINE THE DECODED AND RAW RESULTS
-- =====================================================================
--
-- The final output places the decoded-trade statistics and raw-Swap
-- statistics beside each other for comparison.
--
-- A difference between the two counts would suggest that Dune's
-- standardized dex.trades table does not contain every raw Swap event.
-- =====================================================================

SELECT
    d.pool_name,
    d.pool_address,

    d.number_of_decoded_trades,
    d.first_decoded_trade,
    d.last_decoded_trade,

    r.number_of_raw_swaps,
    r.first_raw_swap,
    r.last_raw_swap

FROM decoded_trades AS d

INNER JOIN raw_swaps AS r
    ON d.pool_address = r.pool_address

ORDER BY d.pool_name;
