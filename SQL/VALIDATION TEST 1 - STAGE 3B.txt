-- ============================================================
-- VALIDATION TEST 1 - STAGE 3B
-- Inspect all ERC-20 token transfers inside the two
-- known wash-trading transactions.
--
-- PURPOSE:
-- Look for transfers to another DEX/protocol that could
-- indicate cross-protocol arbitrage rather than wash trading.
-- ============================================================

SELECT
    evt_block_time,
    evt_block_number,
    evt_tx_hash,
    contract_address AS token_address,
    "from",
    "to",
    value
FROM erc20_ethereum.evt_transfer
WHERE evt_tx_hash IN (
    0x78d1713837aae1d0995383550900ef33efc321e8870bdba7fbf5e398f7484eec,
    0xa28546bd2203675f0dc27cad7a45549f4c71dc1c4cfaa82a485d3eafcc7125ac
)
ORDER BY evt_block_number, evt_index;
